      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (err) {
        console.error("Auth initialization error:", err);
      }
    };
    initAuth();

      setUser(u);
    });
    return () => unsubscribe();

    const unsubscribe = onSnapshot(
      lecturesCol,
      (snapshot) => {
        const docs = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
        if (docs.length === 0) {
          setLectures(INITIAL_LECTURES);
        } else {
          setLectures(docs);
        }
      },
      (error) => {
        console.error("Firestore sync error:", error);
        setLectures(INITIAL_LECTURES);
      }
    );

    return () => unsubscribe();

    e.preventDefault();

          createdAt: new Date().toISOString()

      resetForm();

      console.error("Error saving lecture:", error);
      if (isEditing) {
        setLectures(prev => prev.map(item => item.id === currentLecture.id ? currentLecture : item));

        setLectures(prev => [...prev, { ...currentLecture, id: Date.now().toString() }]);
      }
      setShowModal(false);
      resetForm();

      const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'lectures', id);

                  >
                    <div className="absolute top-0 right-0 w-2 h-full bg-amber-500/40 group-hover:bg-amber-400 transition-all"></div>
                    
                    <div className="flex items-center justify-between mb-6">
                      <span className="text-5xl group-hover:scale-110 transition duration-300">{subject.icon}</span>
                      <span className="px-3 py-1 rounded-full text-xs font-bold bg-amber-500/10 text-amber-300 border border-amber-500/30">
                        سادس علمي
                      </span>
                    </div>

                    <h3 className="text-2xl font-bold text-white group-hover:text-amber-300 transition mb-2">
                      {subject.name}
                    </h3>

                    <p className="text-slate-400 text-xs mb-6 leading-relaxed">
                      {subject.description}
                    </p>

                    <div className="flex items-center justify-between text-sm pt-4 border-t border-slate-800">
                      <span className="text-amber-400 font-semibold flex items-center gap-1.5 text-xs">
                        <FolderOpen className="w-4 h-4" />
                        {count} محاضرة متوفرة
                      </span>
                      <span className="text-xs text-slate-400 group-hover:text-amber-300 transition flex items-center gap-1 font-bold">
                        تصفح الأستاذة
                        <ChevronLeft className="w-4 h-4" />
                      </span>
                    </div>
                  </div>
                );
              })}
            </div>
          </div>
        )}

        {/* PAGE 3: LECTURES LIST & FILTERS */}
        {activeTab === 'lectures' && (
          <div className="space-y-8">
            
            {/* Header & Subject Badge */}
            <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 border-b border-amber-500/20 pb-6">
              <div>
                <h1 className="text-3xl font-extrabold text-white flex items-center gap-3">
                  <GraduationCap className="w-8 h-8 text-amber-400" />
                  المكتبة التعليمية والمحاضرات
                </h1>
                <p className="text-slate-400 text-sm mt-1">
                  {selectedSubject 
                    ? `المحتوى المخصص لمادة: ${SCIENTIFIC_SUBJECTS.find(s => s.id === selectedSubject)?.name}` 
                    : 'استعرض جميع المحاضرات حسب الأستاذ والمادة والفصل'}
                </p>
              </div>

              {selectedSubject && (
                <button
                  onClick={() => setSelectedSubject(null)}
                  className="self-start md:self-auto px-4 py-2 rounded-xl bg-slate-800 text-amber-300 border border-amber-500/30 text-xs font-bold hover:bg-slate-700 transition"
                >
                  عرض جميع المواد السبعة
                </button>
              )}
            </div>

            {/* Filter Controls */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4 bg-[#0b1528] p-4 rounded-2xl border border-amber-500/20">
              
              {/* Search Box */}
              <div className="relative">
                <Search className="absolute right-3.5 top-3.5 w-4 h-4 text-amber-400" />
                <input
                  type="text"
                  placeholder="بحث باسم الأستاذ، عنوان الشرح، أو الفصل..."
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  className="w-full bg-[#060b17] border border-amber-500/20 rounded-xl pr-10 pl-4 py-2.5 text-sm text-white placeholder-slate-500 focus:outline-none focus:border-amber-400"
                />
              </div>

              {/* Subject Select */}
              <select
                value={selectedSubject || ''}
                onChange={(e) => setSelectedSubject(e.target.value || null)}
                className="bg-[#060b17] border border-amber-500/20 rounded-xl px-4 py-2.5 text-sm text-white focus:outline-none focus:border-amber-400"
              >
                <option value="">جميع المواد العلمية</option>
                {SCIENTIFIC_SUBJECTS.map(s => (
                  <option key={s.id} value={s.id}>{s.name}</option>
                ))}
              </select>
            </div>

            {/* Lecture Cards */}
            {filteredLectures.length === 0 ? (
              <div className="text-center py-16 bg-[#0b1528] rounded-3xl border border-slate-800">
                <Video className="w-16 h-16 text-slate-600 mx-auto mb-4" />
                <h3 className="text-xl font-bold text-white mb-2">لا توجد محاضرات تطابق البحث</h3>
                <p className="text-slate-400 text-sm">جرب اختيار مادة أخرى أو مسح كلمة البحث</p>
              </div>
            ) : (
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
                {filteredLectures.map((lec) => {
                  const subjectObj = SCIENTIFIC_SUBJECTS.find(s => s.id === lec.subjectId);
                  const isFav = favorites.includes(lec.id);

                  return (
                    <div
                      key={lec.id}
                      className="rounded-2xl bg-gradient-to-b from-[#0d182e] to-[#070d1a] border border-amber-500/20 hover:border-amber-400/50 p-6 flex flex-col justify-between transition duration-300 hover:shadow-xl hover:shadow-amber-500/5 relative group"
                    >
                      <div>
                        {/* Header Tags & Favorite */}
                        <div className="flex items-center justify-between mb-4">
                          <span className="px-3 py-1 rounded-full text-xs font-bold bg-amber-500/10 text-amber-300 border border-amber-500/20 flex items-center gap-1">
                            <span>{subjectObj?.icon}</span>
                            <span>{subjectObj?.name || 'مادة علمية'}</span>
                          </span>

                          <div className="flex items-center gap-2">
                            <span className="px-2.5 py-0.5 rounded-md text-[11px] font-semibold bg-sky-500/10 text-sky-400 border border-sky-500/20">
                              {lec.type === 'video' ? 'فيديو تليجرام' : lec.type === 'summary' ? 'ملخص' : 'وزاريات'}
                            </span>
                            <button
                              onClick={() => toggleFavorite(lec.id)}
                              className="text-slate-400 hover:text-amber-400 transition"
                            >
                              {isFav ? <BookmarkCheck className="w-5 h-5 text-amber-400 fill-amber-400/20" /> : <Bookmark className="w-5 h-5" />}
                            </button>
                          </div>
                        </div>

                        {/* Teacher Name */}
                        <div className="flex items-center gap-2 text-amber-400 font-bold text-sm mb-2">
                          <User className="w-4 h-4" />
                          <span>{lec.teacherName || 'استاذ المادة'}</span>
                        </div>

                        {/* Chapter */}
                        {lec.chapter && (
                          <div className="text-xs text-sky-300 font-semibold mb-2">
                            {lec.chapter}
                          </div>
                        )}

                        {/* Title */}
                        <h3 className="text-base font-bold text-white group-hover:text-amber-300 transition line-clamp-2 mb-3">
                          {lec.title}
                        </h3>

                        {/* Description */}
                        <p className="text-slate-400 text-xs line-clamp-3 leading-relaxed mb-6">
                          {lec.description || 'لا يوجد وصف إضافي.'}
                        </p>
                      </div>

                      {/* Action Telegram Button */}
                      <div className="space-y-2 pt-2 border-t border-slate-800/80">
                        <a
                          href={lec.telegramUrl}
                          target="_blank"
                          rel="noopener noreferrer"
                          className="w-full py-2.5 rounded-xl bg-gradient-to-r from-sky-500 to-sky-600 hover:from-sky-400 hover:to-sky-500 text-slate-950 font-extrabold text-xs sm:text-sm shadow-md shadow-sky-500/20 transition flex items-center justify-center gap-2"
                        >
                          <Send className="w-4 h-4" />
                          <span>شاهد المحاضرة عبر تليجرام</span>
                          <ExternalLink className="w-3.5 h-3.5 opacity-80" />
                        </a>

                        {lec.pdfUrl && (
                          <a
                            href={lec.pdfUrl}
                            target="_blank"
                            rel="noopener noreferrer"
                            className="w-full py-2 rounded-xl bg-slate-800 hover:bg-slate-700 text-amber-300 font-bold text-xs border border-amber-500/20 transition flex items-center justify-center gap-1.5"
                          >
                            <FileText className="w-3.5 h-3.5" />
                            <span>تحميل الملزمة / PDF</span>
                          </a>
                        )}
                      </div>

                    </div>
                  );
                })}
              </div>
            )}
          </div>
        )}

        {/* PAGE 4: ADMIN DASHBOARD */}
        {activeTab === 'admin' && (
          <div className="max-w-4xl mx-auto space-y-8">
            
            {!isAdminLoggedIn ? (
              <div className="bg-[#0b1528] border border-amber-500/30 rounded-3xl p-8 sm:p-12 text-center shadow-2xl">
                <div className="w-16 h-16 rounded-2xl bg-amber-500/10 border border-amber-500/30 flex items-center justify-center mx-auto mb-6 text-amber-400">
                  <Lock className="w-8 h-8" />
                </div>
                
                <h2 className="text-2xl font-bold text-white mb-2">لوحة الإدارة للفرع العلمي</h2>
                <p className="text-slate-400 text-sm mb-6">أدخل كلمة المرور لإضافة وتعديل الأستاذة والمحاضرات وروابط التليجرام</p>

                <form onSubmit={handleAdminLogin} className="max-w-sm mx-auto space-y-4">
                  <input
                    type="password"
                    placeholder="كلمة المرور (admin)"
                    value={adminPassword}
                    onChange={(e) => setAdminPassword(e.target.value)}
                    className="w-full bg-[#060b17] border border-amber-500/30 rounded-xl px-4 py-3 text-center text-white placeholder-slate-500 focus:outline-none focus:border-amber-400"
                  />
                  {adminError && <p className="text-rose-400 text-xs font-semibold">{adminError}</p>}
                  
                  <button
                    type="submit"
                    className="w-full py-3 rounded-xl bg-gradient-to-r from-amber-400 to-amber-600 text-slate-950 font-bold hover:brightness-110 transition shadow-lg shadow-amber-500/20"
                  >
                    تسجيل الدخول
                  </button>
                </form>
              </div>
            ) : (
              <div className="space-y-8">
                
                {/* Admin Header */}
                <div className="flex flex-col sm:flex-row items-center justify-between gap-4 bg-[#0b1528] p-6 rounded-2xl border border-amber-500/30">
                  <div className="flex items-center gap-3">
                    <div className="p-3 bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 rounded-xl">
                      <CheckCircle className="w-6 h-6" />
                    </div>
                    <div>
                      <h2 className="text-xl font-bold text-white">إدارة المحاضرات والأستاذة</h2>
                      <p className="text-xs text-slate-400">لوحة تحكم المنصة (السادس العلمي)</p>
                    </div>
                  </div>

                  <div className="flex items-center gap-3">
                    <button
                      onClick={() => { resetForm(); setShowModal(true); }}
                      className="px-4 py-2.5 rounded-xl bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold text-xs flex items-center gap-2 shadow-lg shadow-amber-500/20 transition"
                    >
                      <Plus className="w-4 h-4" />
                      إضافة محاضرة جديدة
                    </button>

                    <button
                      onClick={() => setIsAdminLoggedIn(false)}
                      className="p-2.5 rounded-xl bg-slate-800 hover:bg-rose-500/20 text-slate-400 hover:text-rose-400 border border-slate-700 transition"
                      title="تسجيل الخروج"
                    >
                      <LogOut className="w-4 h-4" />
                    </button>
                  </div>
                </div>

                {/* Table / List */}
                <div className="bg-[#0b1528] border border-amber-500/20 rounded-2xl overflow-hidden shadow-xl">
                  <div className="p-4 border-b border-slate-800 font-bold text-white text-sm">
                    المحاضرات المضافة ({lectures.length})
                  </div>

                  <div className="divide-y divide-slate-800/60">
                    {lectures.map((lec) => {
                      const subjectObj = SCIENTIFIC_SUBJECTS.find(s => s.id === lec.subjectId);
                      return (
                        <div key={lec.id} className="p-4 flex flex-col sm:flex-row sm:items-center justify-between gap-4 hover:bg-slate-900/50 transition">
                          <div className="flex items-start gap-3">
                            <span className="text-2xl mt-1">{subjectObj?.icon || '📚'}</span>
                            <div>
                              <h4 className="font-bold text-white text-sm">{lec.title}</h4>
                              <p className="text-xs text-amber-400 font-medium mt-0.5">
                                الأستاذ: {lec.teacherName || 'غير محدد'} | {subjectObj?.name} | {lec.chapter}
                              </p>
                              <p className="text-[11px] text-sky-400 truncate max-w-md mt-1">
                                {lec.telegramUrl}
                              </p>
                            </div>
                          </div>

                          <div className="flex items-center gap-2 self-end sm:self-center">
                            <button
                              onClick={() => openEditModal(lec)}
                              className="p-2 rounded-lg bg-slate-800 hover:bg-amber-500/20 text-amber-400 border border-amber-500/30 transition text-xs flex items-center gap-1"
                            >
                              <Edit3 className="w-3.5 h-3.5" />
                              تعديل
                            </button>

                            <button
                              onClick={() => handleDeleteLecture(lec.id)}
                              className="p-2 rounded-lg bg-slate-800 hover:bg-rose-500/20 text-rose-400 border border-rose-500/30 transition text-xs flex items-center gap-1"
                            >
                              <Trash2 className="w-3.5 h-3.5" />
                              حذف
                            </button>
                          </div>
                        </div>
                      );
                    })}
                  </div>
                </div>

              </div>
            )}

          </div>
        )}

      </main>

      {/* MODAL: Add / Edit Lecture */}
      {showModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/80 backdrop-blur-sm">
          <div className="bg-[#0b1528] border border-amber-500/30 rounded-3xl max-w-lg w-full p-6 space-y-5 shadow-2xl relative">
            <button
              onClick={() => setShowModal(false)}
              className="absolute top-5 left-5 text-slate-400 hover:text-white"
            >
              <X className="w-5 h-5" />
            </button>

            <h3 className="text-xl font-bold text-white border-b border-slate-800 pb-3">
              {isEditing ? 'تعديل بيانات المحاضرة' : 'إضافة محاضرة جديدة'}
            </h3>

            <form onSubmit={handleSaveLecture} className="space-y-4">
              
              {/* Teacher Name */}
              <div>
                <label className="block text-xs font-semibold text-slate-300 mb-1">اسم الأستاذ ⭐</label>
                <input
                  type="text"
                  required
                  placeholder="مثال: الأستاذ حيدر وليد"
                  value={currentLecture.teacherName}
                  onChange={(e) => setCurrentLecture({ ...currentLecture, teacherName: e.target.value })}
                  className="w-full bg-[#060b17] border border-amber-500/20 rounded-xl px-4 py-2.5 text-sm text-white focus:outline-none focus:border-amber-400"
                />
              </div>

              {/* Title */}
              <div>
                <label className="block text-xs font-semibold text-slate-300 mb-1">عنوان المحاضرة</label>
                <input
                  type="text"
                  required
                  placeholder="مثال: إيجاد قيم x, y الحقيقية (أسئلة وزارية)"
                  value={currentLecture.title}
                  onChange={(e) => setCurrentLecture({ ...currentLecture, title: e.target.value })}
                  className="w-full bg-[#060b17] border border-amber-500/20 rounded-xl px-4 py-2.5 text-sm text-white focus:outline-none focus:border-amber-400"
                />
              </div>

              <div className="grid grid-cols-2 gap-4">
                {/* Subject */}
                <div>
                  <label className="block text-xs font-semibold text-slate-300 mb-1">المادة العلميّة</label>
                  <select
                    value={currentLecture.subjectId}
                    onChange={(e) => setCurrentLecture({ ...currentLecture, subjectId: e.target.value })}
                    className="w-full bg-[#060b17] border border-amber-500/20 rounded-xl px-3 py-2.5 text-sm text-white focus:outline-none focus:border-amber-400"
                  >
                    {SCIENTIFIC_SUBJECTS.map(s => (
                      <option key={s.id} value={s.id}>{s.name}</option>
                    ))}
                  </select>
                </div>

                {/* Chapter */}
                <div>
                  <label className="block text-xs font-semibold text-slate-300 mb-1">الفصل / الوحدة</label>
                  <input
                    type="text"
                    placeholder="مثال: الفصل الأول"
                    value={currentLecture.chapter}
                    onChange={(e) => setCurrentLecture({ ...currentLecture, chapter: e.target.value })}
                    className="w-full bg-[#060b17] border border-amber-500/20 rounded-xl px-3 py-2.5 text-sm text-white focus:outline-none focus:border-amber-400"
                  />
                </div>
              </div>

              {/* Telegram Link */}
              <div>
                <label className="block text-xs font-semibold text-sky-400 mb-1">رابط الفيديو / القناة في تليجرام (Telegram Link) 📲</label>
                <input
                  type="url"
                  required
                  placeholder="https://t.me/channel_name/123"
                  value={currentLecture.telegramUrl}
                  onChange={(e) => setCurrentLecture({ ...currentLecture, telegramUrl: e.target.value })}
                  className="w-full bg-[#060b17] border border-sky-500/30 rounded-xl px-4 py-2.5 text-sm text-white focus:outline-none focus:border-sky-400"
                />
              </div>

              {/* PDF Optional Link */}
              <div>
                <label className="block text-xs font-semibold text-slate-300 mb-1">رابط الملزمة أو PDF (اختياري)</label>
                <input
                  type="url"
                  placeholder="https://t.me/channel_name/456"
                  value={currentLecture.pdfUrl}
                  onChange={(e) => setCurrentLecture({ ...currentLecture, pdfUrl: e.target.value })}
                  className="w-full bg-[#060b17] border border-amber-500/20 rounded-xl px-4 py-2.5 text-sm text-white focus:outline-none focus:border-amber-400"
                />
              </div>

              {/* Description */}
              <div>
                <label className="block text-xs font-semibold text-slate-300 mb-1">توضيح أو وصف مختصر</label>
                <textarea
                  rows="2"
                  placeholder="ملاحظات حول الشرح أو الواجبات..."
                  value={currentLecture.description}
                  onChange={(e) => setCurrentLecture({ ...currentLecture, description: e.target.value })}
                  className="w-full bg-[#060b17] border border-amber-500/20 rounded-xl p-3 text-sm text-white focus:outline-none focus:border-amber-400 resize-none"
                ></textarea>
              </div>

              <div className="flex items-center justify-end gap-3 pt-2">
                <button
                  type="button"
                  onClick={() => setShowModal(false)}
                  className="px-4 py-2 rounded-xl text-slate-400 hover:text-white text-xs font-semibold"
                >
                  إلغاء
                </button>
                <button
                  type="submit"
                  className="px-6 py-2.5 rounded-xl bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold text-xs shadow-lg shadow-amber-500/20"
                >
                  حفظ المحاضرة
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Footer */}
      <footer className="mt-20 border-t border-amber-500/20 py-8 bg-[#040812]">
        <div className="max-w-7xl mx-auto px-4 text-center space-y-2">
          <div className="flex items-center justify-center gap-2 text-amber-400 font-bold">
            <GraduationCap className="w-5 h-5" />
            منصة «مِنبَر العلم 🎓» - الفرع العلمي
          </div>
          <p className="text-slate-500 text-xs">
            جميع الحقوق محفوظة © {new Date().getFullYear()} - مخصصة لخدمة طلبة السادس العلمي في العراق
          </p>
        </div>
      </footer>

    </div>
  );
}
